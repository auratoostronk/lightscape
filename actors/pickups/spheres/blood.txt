//------------------------------------------------------------
//Pickup class for Blood Sphere
//------------------------------------------------------------
Actor BloodSpherePickup : CustomInventory 12102
{
	//Doom Builder things category
	//$Category Lightscape Pickups
	
	//Propreties
	Radius 24
	Height 48
	Inventory.Amount 1
	Inventory.MaxAmount 1
	Inventory.PickupMessage "Blood Sphere - Fires a projectile that siphons health."
	Inventory.PickupSound "Pickups/Sphere"
	Inventory.RespawnTics 700 // 20 seconds
	//Scale 0.5
	Tag "Blood Sphere"
	
	//Flags
	+FLOATBOB
	+BRIGHT
	+SERVERNETID
	+INVBAR
	
	//States
	States
	{
	Spawn:
		SPHP C 1
		loop
	Pickup:
		TNT1 A 0 A_JumpIf(CallACS("CheckForSphere") == 0, "Success") //Only allow pickup if no other active items are held.
		fail
	Success:
		TNT1 A 0 ACS_NamedExecuteAlways("HoldingSphere", 0, 0)
		TNT1 A 0 A_GiveInventory("BloodSphereReady", 1)
		stop
	}
}

//------------------------------------------------------------
//Blood Sphere projectile
//------------------------------------------------------------
ACTOR BloodSphereProjectile
{
	//Properties
	Radius 6
	Height 12
	Speed 48
	Damage (0)
	Projectile
	RenderStyle Add
	Gravity 0.5
	SeeSound "Items/BloodSphere/Fire"
	Obituary "%k > \chBlood Sphere\c- > %o"
	
	//Flags
	-NOGRAVITY
	+BRIGHT
	+SEEKERMISSILE //For hit sounds
	
	//Constants
	const int RADIUS_OUTER = 160;
	
	//States
	States
	{
	Spawn:
		SPHR C 1 A_SpawnItemEx("BloodSphereProjectileTrail",0,0,0,0,0,0,0,SXF_NOCHECKPOSITION)
		loop
	XDeath:
		TNT1 A 0 A_SpawnItemEx("ExplosionSpawner")
		TNT1 AAAAAAAAAAAAAAAA 0 A_SpawnItemEx("BloodSphereImpactFX", 0, 0, 0, frandom(-16.0, 16.0), frandom(-16.0, 16.0), frandom(-16.0, 16.0), 0, SXF_NOCHECKPOSITION)
		TNT1 A 0 A_Quake(2, 8, 0, 384)
		TNT1 A 0 A_Explode(20, RADIUS_OUTER, XF_HURTSOURCE)
		TNT1 A 0 A_Explode(20, RADIUS_OUTER, XF_HURTSOURCE, true, RADIUS_OUTER)
		TNT1 A 0 ACS_NamedExecuteAlways("WeaponHit", 0, 2)
		TNT1 AAAAAAAAAA 2 A_CustomMissile("HealingFXSeeker", frandom(-8.0, 8.0), frandom(-8.0, 8.0), 0, CMF_TRACKOWNER, frandom(-8.0, 8.0))
		TNT1 AAAAAAAAAAAAAAAAAAAAAAAAAAAAAA 2 A_GiveToTarget("Health", 1) //30 Health
		Goto FinishDeath
	Death:
		TNT1 A 0 A_SpawnItemEx("ExplosionSpawner")
		TNT1 AAAAAAAAAAAAAAAA 0 A_SpawnItemEx("BloodSphereImpactFX", 0, 0, 0, frandom(-16.0, 16.0), frandom(-16.0, 16.0), frandom(-16.0, 16.0), 0, SXF_NOCHECKPOSITION)
		TNT1 A 0 A_Quake(2, 8, 0, 384)
		TNT1 A 0 A_Explode(20, RADIUS_OUTER, XF_HURTSOURCE)
		TNT1 A 0 A_Explode(20, RADIUS_OUTER, XF_HURTSOURCE, true, RADIUS_OUTER)
		TNT1 A 0 A_SeekerMissile(360, 360, SMF_LOOK, 256, RADIUS_OUTER)
		TNT1 A 0 A_RearrangePointers(AAPTR_DEFAULT, AAPTR_TRACER)
		TNT1 A 0 A_JumpIfMasterCloser(RADIUS_OUTER - radius, "HitSoft")
		Goto FinishDeath
	HitSoft:
		TNT1 A 0 ACS_NamedExecuteAlways("WeaponHit", 0, 0)
		TNT1 AAAAAAAAAA 2 A_CustomMissile("HealingFXSeeker", frandom(-8.0, 8.0), frandom(-8.0, 8.0), 0, CMF_TRACKOWNER, frandom(-8.0, 8.0))
		TNT1 AAAAAAAAAAAAAAAAAAAAAAAAAAAAAA 2 A_GiveToTarget("Health", 1) //30 Health
		Goto FinishDeath
	FinishDeath:
		TNT1 A 105 //Required for unlagged
		stop
	}
}
