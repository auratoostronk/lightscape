//------------------------------------------------------------
//Flamespitter weapon class
//------------------------------------------------------------
ACTOR Flamespitter : LSWeaponBase
{
	//Properties
	Tag "Flamespitter"
	Weapon.AmmoType1 "FlameAmmo"
	Weapon.AmmoType2 "FlameAmmo"
	Weapon.AmmoUse1 1
	Weapon.AmmoUse2 10
	Weapon.SlotNumber 1
	Inventory.Icon "CARPA0"

	//States
	States
	{
	Spawn:
		FLMP A 1 bright A_SetAngle(angle + 1.0)
		loop
		
	Ready:
		TNT1 A 0 A_StopSound(CHAN_WEAPON)
		FLMR A 1 A_WeaponReady
		Loop
		
	Deselect:
		TNT1 A 0 A_ZoomFactor(1.0)
		TNT1 A 0 A_StopSound(CHAN_WEAPON)
		TNT1 AA 0 A_Lower
		FLMR A 1 A_Lower
		Loop
		
	Select:
		TNT1 AA 0 A_Raise
		FLMR A 1 A_Raise
		Loop
		
	Fire:
		FLMR A 6 A_PlayWeaponSound("Flamespitter/FlameStart")
		TNT1 A 0 A_PlaySound("Flamespitter/FlameLoop", CHAN_WEAPON, 1.0, true)
		goto HoldNormal
		
	HoldNormal:
		TNT1 A 0 A_GiveInventory("MainAmmoRegenCD", 999)
		TNT1 A 0 A_Jump(256, "HoldNormalLeft", "HoldNormalUp", "HoldNormalRight", "HoldNormalDown")
		Goto HoldNormalLeft
		
	HoldNormalLeft:
		FLMF B 1 offset(0, 24) A_FireCustomMissile("FlamespitterProjectile", afrandom(-2.0, 2.0), true, 3.0, 5.0, 0, afrandom(-2.0, 2.0))
		TNT1 A 0 A_ReFire("HoldNormal")
		Goto FireStop
		
	HoldNormalUp:
		FLMF B 1 offset(8, 16) A_FireCustomMissile("FlamespitterProjectile", afrandom(-2.0, 2.0), true, 3.0, 5.0, 0, afrandom(-2.0, 2.0))
		TNT1 A 0 A_ReFire("HoldNormal")
		Goto FireStop
		
	HoldNormalRight:
		FLMF B 1 offset(16, 24) A_FireCustomMissile("FlamespitterProjectile", afrandom(-2.0, 2.0), true, 3.0, 5.0, 0, afrandom(-2.0, 2.0))
		TNT1 A 0 A_ReFire("HoldNormal")
		Goto FireStop
		
	HoldNormalDown:
		FLMF B 1 offset(8, 32) A_FireCustomMissile("FlamespitterProjectile", afrandom(-2.0, 2.0), true, 3.0, 5.0, 0, afrandom(-2.0, 2.0))
		TNT1 A 0 A_ReFire("HoldNormal")
		Goto FireStop
		
	FireStop:
		FLMF C 2 A_GiveInventory("MainAmmoRegenCD", 999)
		TNT1 A 0 A_StopSound(CHAN_WEAPON)
		FLMF D 2 A_PlayWeaponSound("Flamespitter/FlameStop")
		FLMF EFGHIJKLLKJIHGFE 1
		Goto Ready
		
	AltFire:
		TNT1 A 0 A_JumpIfInventory("MainAltCD", 1, "AltFireFail")
		TNT1 A 0 A_GiveInventory("MainAmmoRegenCD", 999)
		TNT1 A 0 A_GiveInventory("MainAltCD", 8)
		TNT1 A 0 A_SpawnItemEX("OrangeLightShoot")
		TNT1 A 0 A_PlaySound("C14/AltFire", CHAN_WEAPON)
		//TNT1 A 0 A_PlaySound("Flamespitter/AltFireFar", CHAN_6)
		TNT1 A 0 A_Recoil(4)
		TNT1 A 0 Bright A_FireCustomMissile("FlamespitterCanister", 0.0, true, 2.0, 5.0)
		FLMF EFGHIJKLLKJIHGFE 1
		Goto Ready
		
	AltFireFail:
		TNT1 A 0 ACS_NamedExecuteAlways("AltFireFail")
		FLMR A 1 A_WeaponReady(WRF_NOSECONDARY)
		Goto Ready
		
	}
}

//------------------------------------------------------------
//Flamespitter flame projectile
//------------------------------------------------------------
Actor FlamespitterProjectile
{
	//Properties
	Projectile
	Speed 40
	Radius 8
	Height 12
	Damage (3)
	Scale 0.25
	Renderstyle Add
	Alpha 0.3
	DamageType "FlamespitterFlames"
	Obituary "%k > \ciFlamespitter\c- > %o"
	Decal "Scorch"
	
	//Flags
	+BRIGHT
	+DONTSPLASH
	+RIPPER
	
	//Flame projectile velocity slow amount.
	const float flame_slowscale = 0.7;
	
	//States
	States
	{
	Spawn:
		OFLM A 4 nodelay A_SetScale(scalex + frandom(-0.05, 0.05), scaley + frandom(-0.05, 0.05))
		TNT1 A 0 A_SetScale(scalex + 0.1, scaley + 0.1)
		OFLM B 4 A_ChangeVelocity(velx * flame_slowscale, vely * flame_slowscale, velz * flame_slowscale, CVF_REPLACE)
		TNT1 A 0 A_SetScale(scalex + 0.1, scaley + 0.1)
		OFLM C 4 A_ChangeVelocity(velx * flame_slowscale, vely * flame_slowscale, velz * flame_slowscale, CVF_REPLACE)
		TNT1 A 0 A_SetScale(scalex + 0.1, scaley + 0.1)
		OFLM D 4 A_ChangeVelocity(velx * flame_slowscale, vely * flame_slowscale, velz * flame_slowscale, CVF_REPLACE)
		TNT1 A 0 A_SetScale(scalex + 0.1, scaley + 0.1)
		OFLM E 4 A_ChangeVelocity(velx * flame_slowscale, vely * flame_slowscale, velz * flame_slowscale, CVF_REPLACE)
		TNT1 A 0 A_SetScale(scalex + 0.1, scaley + 0.1)
		OFLM F 4 A_ChangeVelocity(velx * flame_slowscale, vely * flame_slowscale, velz * flame_slowscale, CVF_REPLACE)
		TNT1 A 0 A_SetScale(scalex + 0.1, scaley + 0.1)
		OFLM G 4 A_ChangeVelocity(velx * flame_slowscale, vely * flame_slowscale, velz * flame_slowscale, CVF_REPLACE)
		Goto Death
	Death:
		TNT1 A 35
		stop
	}
}
//Canister blast radii
const int CANISTER_RANGE_OUTER = 192;
const int CANISTER_RANGE_INNER = 64;

//------------------------------------------------------------
//Flamespitter alt-fire canister
//------------------------------------------------------------
Actor FlamespitterCanister
{
	//Properties
	Projectile
	Speed 48
	Radius 8
	Height 12
	Damage (0)  
	Scale 0.8
	Obituary "%k > \ciFlamespitter\c- > %o"
	DamageType "FlamespitterAlt"
	Decal "PlasmaScorchLower"
	
	//Flags
	+BRIGHT
	-NOGRAVITY
	
	//States
	States
	{
	Spawn:
		FCAN A 16
		Goto Death

	XDeath:
		TNT1 A 0 A_GiveToTarget("HitSoundLarge", 1)
		goto Death

	Death:
		TNT1 A 0 A_SpawnItemEx("ExplosionSpawner")
		TNT1 A 0 A_SpawnItemEX("GasolineSplash")
		TNT1 A 0 A_Quake(2.0, 6, 0, CANISTER_RANGE_OUTER * 2.0)
		TNT1 A 0 A_Explode(50, CANISTER_RANGE_OUTER, 0, true, CANISTER_RANGE_INNER)
		TNT1 A 0 A_SeekerMissile(360, 360, SMF_LOOK, 256, CANISTER_RANGE_OUTER / 128)
		TNT1 A 0 A_RearrangePointers(AAPTR_DEFAULT, AAPTR_TRACER)
		TNT1 A 0 A_JumpIfMasterCloser(CANISTER_RANGE_INNER + radius, "HitMedium")
		TNT1 A 0 A_JumpIfMasterCloser(CANISTER_RANGE_OUTER + radius, "HitSoft")
		TNT1 A 105  //Required for unlagged
		stop

	HitMedium:
		TNT1 A 0 A_GiveToTarget("HitSoundMedium", 1)
		TNT1 A 105  //Required for unlagged
		stop

	HitSoft:
		TNT1 A 0 A_GiveToTarget("HitSoundSmall", 1)
		TNT1 A 105  //Required for unlagged
		stop
		
	}
}
