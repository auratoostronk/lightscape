//------------------------------------------------------------
//Beam weapon class
//------------------------------------------------------------
ACTOR BeamRifle : LSWeaponBase
{
	//Properties
	Tag "Beam Rifle"
	Weapon.AmmoType1 "BeamAmmo"
	Weapon.AmmoType2 "BeamAmmo"
	Weapon.AmmoUse1 1
	Weapon.AmmoUse2 2
	Weapon.SlotNumber 1
	Inventory.Icon "BRFPA0"
	
	Decal "BeamMainDecal"

	//States
	States
	{
	Spawn:
		BRFP A 1 bright A_SetAngle(angle + 1.0)
		loop

	Ready:
		BRFR A 1 A_WeaponReady
		loop

	Select:
		BRFR AA 0 A_Raise
		BRFR A 1 A_Raise
		loop

	Deselect:
		TNT1 A 0 A_ZoomFactor(1.0)
		TNT1 A 0 A_StopSound(CHAN_WEAPON)	//These are here in case ammo runs out or the player dies while firing
		TNT1 A 0 A_StopSound(CHAN_7)		//These are here in case ammo runs out or the player dies while firing
		BRFR AA 0 A_Lower
		BRFR A 1 A_Lower
		loop
		
	Fire:
		TNT1 A 0 A_SpawnItemEX("LightBlueLightShoot")
		TNT1 A 0 A_PlaySound("BeamRifle/Fire", CHAN_WEAPON, 1.0, true)
		TNT1 A 0 A_PlaySound("BeamRifle/FireFar", CHAN_7, 1.0, true)
		TNT1 A 0 A_Refire("HoldNormal")
		Goto Ready
	HoldNormal:
		TNT1 A 0 A_GiveInventory("MainAmmoRegenCD", 999)
		TNT1 A 0 A_Jump(256, "HoldNormalLeft", "HoldNormalUp", "HoldNormalRight", "HoldNormalDown")
	HoldNormalLeft:
		BRFR A 1 offset(-8, 32) A_RailAttack(3, 2.5, true, "none", "none", RGF_SILENT, 0, "BeamHitPuff", 0, 0, 720.0, 1, 16.0, 0, "BeamRailTrail", 9.0)
		TNT1 A 0 A_ReFire("HoldNormal")
		Goto FireStop
	HoldNormalUp:
		BRFR A 1 offset(0, 24) A_RailAttack(3, 2.5, true, "none", "none", RGF_SILENT, 0, "BeamHitPuff", 0, 0, 720.0, 1, 16.0, 0, "BeamRailTrail", 9.0)
		TNT1 A 0 A_ReFire("HoldNormal")
		Goto FireStop
	HoldNormalRight:
		BRFR A 1 offset(8, 32) A_RailAttack(3, 2.5, true, "none", "none", RGF_SILENT, 0, "BeamHitPuff", 0, 0, 720.0, 1, 16.0, 0, "BeamRailTrail", 9.0)
		TNT1 A 0 A_ReFire("HoldNormal")
		Goto FireStop
	HoldNormalDown:
		BRFR A 1 offset(0, 40) A_RailAttack(3, 2.5, true, "none", "none", RGF_SILENT, 0, "BeamHitPuff", 0, 0, 720.0, 1, 16.0, 0, "BeamRailTrail", 9.0)
		TNT1 A 0 A_ReFire("HoldNormal")
		Goto FireStop
	FireStop:
		BRFF FG 1
		TNT1 A 0 A_StopSound(CHAN_WEAPON)	//These are here in case ammo runs out or the player dies while firing
		TNT1 A 0 A_StopSound(CHAN_7)		//These are here in case ammo runs out or the player dies while firing
		BRFR A 6 A_WeaponReady
		Goto Ready
		
	AltFire:
		TNT1 A 0 A_GiveInventory("MainAmmoRegenCD", 999)
		TNT1 A 0 A_SpawnItemEX("LightBlueLightShoot")
		TNT1 A 0 A_PlaySound("BeamRifle/AltFire", CHAN_WEAPON, 1.0, false)
		TNT1 A 0 A_PlaySound("BeamRifle/AltFireFar", CHAN_7, 1.0, false)
		BRFF C 1 A_RailAttack(13, 2.5, true, "none", "none", RGF_SILENT, 0, "BeamAltHitPuff", 0, 0, 16384.0, 1, 8.0, 0, "BeamAltRailTrail", 9.0)
		BRFF DEFG 1
		Goto Ready
		
	AltFireFail:
		TNT1 A 0 ACS_NamedExecuteAlways("AltFireFail")
		BRFR A 1 A_WeaponReady(WRF_NOPRIMARY|WRF_NOSECONDARY)
		Goto Ready
	}
}
