//------------------------------------------------------------
//Explosive Barrel class
//------------------------------------------------------------
ACTOR LSBarrelExplosive : ExplosiveBarrel replaces ExplosiveBarrel
{
	//Doom Builder things category
	//$Category Lightscape Hazards
	Health 40
	Radius 16
	Height 40
	Mass 20
	PainChance 255
	+SOLID
	+SHOOTABLE
	+NOBLOOD
	+ACTIVATEMCROSS
	+NOICEDEATH
	Obituary "%k > Barrel > %o"
	States
	{
	Spawn:
		3DOB A 1
		Loop
	Pain:
		TNT1 A 0 A_SpawnItemEX("LSBarrelExplosiveFlames", 0, 0, 32.0, frandom(-4.0, 4.0), frandom(-4.0, 4.0), frandom(4.0, 8.0))
		3DOB A 2 
		Goto Ticking
	Ticking:
		TNT1 A 0 A_SpawnItemEX("LSBarrelExplosiveFlames", 0, 0, 32.0, frandom(-4.0, 4.0), frandom(-4.0, 4.0), frandom(4.0, 8.0))
		3DOB A 2 A_SetArg(0, args[0] + 1)
		TNT1 A 0 A_JumpIf(args[0] > 35, "Death")
		loop
	Death:
		TNT1 A 0 A_SpawnItemEX("LSBarrelExplosiveDebris1", 0, 0, 0, frandom(-16.0, 16.0), frandom(-16.0, 16.0), frandom(-16.0, 16.0))
		TNT1 AAAAAAAA 0 A_SpawnItemEX("LSBarrelExplosiveFlamesSpawner", 0, 0, 0, frandom(-8.0, 8.0), frandom(-8.0, 8.0), frandom(-8.0, 8.0))
		TNT1 A 0 A_SpawnItemEx("ExplosionSpawner")
		TNT1 A 5 Bright A_Explode(60, 192, XF_HURTSOURCE, true, 96)
		TNT1 A 10 Bright
		TNT1 A 1050 Bright A_BarrelDestroy
		TNT1 A 5 A_Respawn
		Wait
	}
}

//------------------------------------------------------------
//Explosive Barrel Debris
//------------------------------------------------------------
ACTOR LSBarrelExplosiveDebris1 : LSBouncingParticleBase
{
	RenderStyle Normal
	args 0, 0, 0
	States
	{
	Spawn:
		TNT1 A 0 nodelay A_SetArg(0, frandom(0.0, 32.0))
		TNT1 A 0 A_SetArg(1, frandom(0.0, 32.0))
		//TNT1 A 0 A_SetArg(2, frandom(0.0, 32.0))
	Spawn1:
		TNT1 A 0 A_SetAngle(angle + args[0])
		TNT1 A 0 A_SetPitch(pitch + args[1])
		//TNT1 A 0 A_SetRoll(roll + args[2])
		3DOB A 1
		loop
	Death:
		3DOB A 1 A_FadeOut(0.05)
		loop
	}
}
